# AI Notify Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a 5-NeoPixel + LCD1602 ambient display for the nakom.is AI chat, driven by MQTT events from AWS IoT Core published by the nakom.is stream-handler Lambda.

**Architecture:** Three components: (1) a CDK stack in `ai-notify/infra/` provisions an IoT Thing + device certificate + IAM publish policy; (2) the nakom.is stream-handler Lambda is modified to fire-and-forget MQTT events to `ai-notify/events`; (3) the ESP32-S3-CAM firmware subscribes via PubSubClient, drives NeoPixel LEDs and an I2C LCD1602 through an `AiNotifyController` state machine serviced by two FreeRTOS tasks.

**Tech Stack:** PlatformIO/Arduino (ESP32-S3-CAM), Adafruit NeoPixel, LiquidCrystal_I2C, PubSubClient, ArduinoJson v7, AWS CDK v2 (TypeScript), `cdk-iot-core-certificates-v3`, `@aws-sdk/client-iot-data-plane` (Lambda Node 20 built-in)

```typescript
import * as cdk from 'aws-cdk-lib';
import * as iam from 'aws-cdk-lib/aws-iam';
import * as iot from 'aws-cdk-lib/aws-iot';
import * as ssm from 'aws-cdk-lib/aws-ssm';
import { Construct } from 'constructs';
import { ThingWithCert } from 'cdk-iot-core-certificates-v3';

export class AiNotifyStack extends cdk.Stack {
    constructor(scope: Construct, id: string, props?: cdk.StackProps) {
        super(scope, id, props);

        const projectName = 'AiNotify';
        const thingName   = `${projectName}Thing`;     // AiNotifyThing
        const topic       = 'ai-notify/events';

        // IoT Thing + certificate.
        // ThingWithCert saves to SSM automatically:
        //   /AiNotify/AiNotifyThing/certPem
        //   /AiNotify/AiNotifyThing/privKey
        const { certId } = new ThingWithCert(this, 'Thing', {
            thingName,
            saveToParamStore: true,
            paramPrefix: projectName,
        });

        // IoT policy: device may only connect, subscribe, and receive on the event topic.
        // The device never publishes — that is the Lambda's job.
        const iotPolicy = new iot.CfnPolicy(this, 'DevicePolicy', {
            policyName: `${projectName}DevicePolicy`,
            policyDocument: {
                Version: '2012-10-17',
                Statement: [
                    {
                        Effect:   'Allow',
                        Action:   'iot:Connect',
                        Resource: `arn:aws:iot:${this.region}:${this.account}:client/${thingName}`,
                    },
                    {
                        Effect:   'Allow',
                        Action:   'iot:Subscribe',
                        Resource: `arn:aws:iot:${this.region}:${this.account}:topicfilter/${topic}`,
                    },
                    {
                        Effect:   'Allow',
                        Action:   'iot:Receive',
                        Resource: `arn:aws:iot:${this.region}:${this.account}:topic/${topic}`,
                    },
                ],
            },
        });

        // Attach the device policy to the certificate.
        const certArn = `arn:aws:iot:${this.region}:${this.account}:cert/${certId}`;
        const policyAttachment = new iot.CfnPolicyPrincipalAttachment(this, 'PolicyAttachment', {
            policyName: iotPolicy.policyName!,
            principal:  certArn,
        });
        policyAttachment.addDependency(iotPolicy);

        // IAM Managed Policy for the nakom.is Lambda to publish to the event topic
        // and read the IoT endpoint from SSM.
        const publishPolicy = new iam.ManagedPolicy(this, 'LambdaPublishPolicy', {
            managedPolicyName: 'AiNotifyLambdaPublishPolicy',
            description: 'Allows publishing MQTT events to ai-notify/events and reading the IoT endpoint from SSM',
            statements: [
                new iam.PolicyStatement({
                    effect:    iam.Effect.ALLOW,
                    actions:   ['iot:Publish'],
                    resources: [`arn:aws:iot:${this.region}:${this.account}:topic/${topic}`],
                }),
                new iam.PolicyStatement({
                    effect:    iam.Effect.ALLOW,
                    actions:   ['ssm:GetParameter'],
                    resources: [
                        `arn:aws:ssm:${this.region}:${this.account}:parameter/AiNotify/IotEndpoint`,
                    ],
                }),
            ],
        });

        // Store the policy ARN in SSM so the nakom.is CDK stack can reference it
        // without hardcoding the ARN.
        new ssm.StringParameter(this, 'IotPublishPolicyArn', {
            parameterName: `/${projectName}/IotPublishPolicyArn`,
            stringValue:   publishPolicy.managedPolicyArn,
            description:   'ARN of AiNotifyLambdaPublishPolicy — attach to nakom.is stream Lambda',
        });

        new cdk.CfnOutput(this, 'ThingName',         { value: thingName });
        new cdk.CfnOutput(this, 'PublishPolicyArn',  {
            value:       publishPolicy.managedPolicyArn,
            description: 'Attach this to the nakomis-chat-stream Lambda execution role',
        });
        new cdk.CfnOutput(this, 'SSMCertPath',    { value: `/${projectName}/${thingName}/certPem` });
        new cdk.CfnOutput(this, 'SSMPrivKeyPath', { value: `/${projectName}/${thingName}/privKey` });
    }
}
```

```cpp
#pragma once

#include <Arduino.h>
#include <Adafruit_NeoPixel.h>
#include <LiquidCrystal_I2C.h>
#include <freertos/FreeRTOS.h>
#include <freertos/semphr.h>

#define NEOPIXEL_PIN        48
#define NEOPIXEL_COUNT      5
#define NEOPIXEL_BRIGHTNESS 128   // 50% — prevents eye-searing white on LED2

#define LCD_ADDR  0x27
#define LCD_COLS  16
#define LCD_ROWS  2

// NeoPixel indices
#define LED_CV       0
#define LED_LINKEDIN 1
#define LED_GITHUB   2
#define LED_BLOG     3
#define LED_AI       4

enum class AppState {
    IDLE,
    ACTIVE,       // AI processing, LED4 rainbow cycling
    TOOL_ACTIVE,  // Tool LED on + LED4 still cycling
};

// Returns LED index (0–3) for a tool name, or -1 if not mapped
int toolToLed(const String& tool);

// Returns LCD row-2 status text for a tool_start event
String toolToStatus(const String& tool);

class AiNotifyController {
public:
    AiNotifyController();
    void begin();

    // Called from the MQTT callback (Arduino task context)
    void onQuestion(const String& text);
    void onAiStart();
    void onToolStart(const String& tool);
    void onToolEnd(const String& tool);
    void onDone();
    void onWifiReconnecting();

    // Called from FreeRTOS tasks
    void tickRainbow();   // call every ~50 ms
    void tickMarquee();   // call every ~400 ms

    AppState getState() const { return _state; }

    // Shared mutex — must be held before touching strip or lcd
    SemaphoreHandle_t displayMutex;

private:
    Adafruit_NeoPixel _strip;
    LiquidCrystal_I2C _lcd;

    AppState _state;
    bool     _toolLedOn[4];
    uint8_t  _rainbowIndex;
    String   _currentQuestion;
    String   _statusText;
    int      _marqueeOffset;
    bool     _lcdAvailable;

    void setToolLed(int led, bool on);
    void updateLcdRow2();
    uint32_t rainbowColor(uint8_t pos);
};

// FreeRTOS task entry points — pass AiNotifyController* as param
void ledRainbowTask(void* param);
void lcdMarqueeTask(void* param);
```

**Step 2:** Create `embedded/src/AiNotifyController.cpp`:
```cpp
#include "AiNotifyController.h"
#include "SDLogger.h"

// Tool LED colours at 100% brightness; global setBrightness(128) halves them
static const uint32_t TOOL_COLOURS[4] = {
    Adafruit_NeoPixel::Color(255, 140,   0),   // CV: dark orange
    Adafruit_NeoPixel::Color(  0, 119, 181),   // LinkedIn: blue
    Adafruit_NeoPixel::Color(240, 240, 240),   // GitHub: near-white
    Adafruit_NeoPixel::Color( 56, 152,  63),   // Blog: green
};

int toolToLed(const String& tool) {
    if (tool == "get_cv")                                                  return LED_CV;
    if (tool == "get_linkedin_profile")                                    return LED_LINKEDIN;
    if (tool == "get_github_readme" ||
        tool == "list_repo_files"   ||
        tool == "read_repo_file")                                          return LED_GITHUB;
    if (tool == "get_blog_posts" || tool == "get_interests")               return LED_BLOG;
    return -1;
}

String toolToStatus(const String& tool) {
    if (tool == "get_cv")                                                  return "Searching CV...";
    if (tool == "get_linkedin_profile")                                    return "Searching LinkedIn";
    if (tool == "get_github_readme" ||
        tool == "list_repo_files"   ||
        tool == "read_repo_file")                                          return "Searching GitHub";
    if (tool == "get_blog_posts" || tool == "get_interests")               return "Searching Blog...";
    return "Working...";
}

AiNotifyController::AiNotifyController()
    : _strip(NEOPIXEL_COUNT, NEOPIXEL_PIN, NEO_GRB + NEO_KHZ800),
      _lcd(LCD_ADDR, LCD_COLS, LCD_ROWS),
      _state(AppState::IDLE),
      _toolLedOn{false, false, false, false},
      _rainbowIndex(0),
      _currentQuestion("nakom.is"),
      _statusText("Ready"),
      _marqueeOffset(0),
      _lcdAvailable(false),
      displayMutex(nullptr)
{}

void AiNotifyController::begin() {
    displayMutex = xSemaphoreCreateMutex();

    _strip.begin();
    _strip.setBrightness(NEOPIXEL_BRIGHTNESS);
    _strip.clear();
    _strip.show();

    Wire.begin(8, 9);   // SDA=GPIO8, SCL=GPIO9
    _lcd.init();
    _lcdAvailable = true;
    _lcd.backlight();
    _lcd.clear();
    _lcd.setCursor(0, 0);
    _lcd.print("nakom.is");
    _lcd.setCursor(0, 1);
    _lcd.print("Ready");

    LOG_I("AiNotifyController: ready");
}

void AiNotifyController::onQuestion(const String& text) {
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    _currentQuestion = text;
    _marqueeOffset   = 0;
    _statusText      = "Thinking...";
    _state           = AppState::ACTIVE;
    updateLcdRow2();
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::onAiStart() {
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    _statusText = "Thinking...";
    if (_state == AppState::IDLE) _state = AppState::ACTIVE;
    updateLcdRow2();
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::onToolStart(const String& tool) {
    int led = toolToLed(tool);
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    if (led >= 0 && led < 4) setToolLed(led, true);
    _statusText = toolToStatus(tool);
    _state      = AppState::TOOL_ACTIVE;
    updateLcdRow2();
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::onToolEnd(const String& tool) {
    int led = toolToLed(tool);
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    if (led >= 0 && led < 4) setToolLed(led, false);
    bool anyOn = false;
    for (int i = 0; i < 4; i++) anyOn |= _toolLedOn[i];
    _state = anyOn ? AppState::TOOL_ACTIVE : AppState::ACTIVE;
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::onDone() {
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    for (int i = 0; i < 4; i++) {
        _toolLedOn[i] = false;
        _strip.setPixelColor(i, (uint32_t)0);
    }
    _strip.setPixelColor(LED_AI, (uint32_t)0);
    _strip.show();
    _state      = AppState::IDLE;
    _statusText = "Ready";
    updateLcdRow2();
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::onWifiReconnecting() {
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(100)) != pdTRUE) return;
    _statusText = "Reconnecting...";
    updateLcdRow2();
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::setToolLed(int led, bool on) {
    _toolLedOn[led] = on;
    _strip.setPixelColor(led, on ? TOOL_COLOURS[led] : (uint32_t)0);
    _strip.show();
}

void AiNotifyController::updateLcdRow2() {
    if (!_lcdAvailable) return;
    _lcd.setCursor(0, 1);
    String s = _statusText;
    while ((int)s.length() < LCD_COLS) s += ' ';   // pad to clear old text
    _lcd.print(s.substring(0, LCD_COLS));
}

void AiNotifyController::tickRainbow() {
    if (_state == AppState::IDLE) return;
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(10)) != pdTRUE) return;
    _strip.setPixelColor(LED_AI, rainbowColor(_rainbowIndex));
    _strip.show();
    _rainbowIndex += 3;   // wraps naturally at uint8_t overflow (256)
    xSemaphoreGive(displayMutex);
}

void AiNotifyController::tickMarquee() {
    if (!_lcdAvailable) return;

    // Capture question atomically, then release mutex before slow LCD writes
    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(10)) != pdTRUE) return;
    String q = _currentQuestion;
    xSemaphoreGive(displayMutex);

    String display;
    if ((int)q.length() <= LCD_COLS) {
        // Short question: static display, centred-ish by left-padding optional
        display = q;
        while ((int)display.length() < LCD_COLS) display += ' ';
    } else {
        // Long question: scroll a 16-char window
        String padded = q + "    ";   // 4-space gap before wrap-around
        int len = padded.length();
        display = "";
        for (int i = 0; i < LCD_COLS; i++) {
            display += padded.charAt((_marqueeOffset + i) % len);
        }
        _marqueeOffset = (_marqueeOffset + 1) % len;
    }

    if (xSemaphoreTake(displayMutex, pdMS_TO_TICKS(10)) != pdTRUE) return;
    if (_lcdAvailable) {
        _lcd.setCursor(0, 0);
        _lcd.print(display);
    }
    xSemaphoreGive(displayMutex);
}

uint32_t AiNotifyController::rainbowColor(uint8_t pos) {
    // Standard HSV wheel: Red(0) → Yellow → Green(85) → Cyan → Blue(170) → Purple → Red(255)
    if (pos < 85) {
        return Adafruit_NeoPixel::Color(255 - pos * 3,  pos * 3,        0);
    } else if (pos < 170) {
        pos -= 85;
        return Adafruit_NeoPixel::Color(0,              255 - pos * 3,  pos * 3);
    } else {
        pos -= 170;
        return Adafruit_NeoPixel::Color(pos * 3,        0,              255 - pos * 3);
    }
}

void ledRainbowTask(void* param) {
    AiNotifyController* ctrl = static_cast<AiNotifyController*>(param);
    for (;;) {
        ctrl->tickRainbow();
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

void lcdMarqueeTask(void* param) {
    AiNotifyController* ctrl = static_cast<AiNotifyController*>(param);
    for (;;) {
        ctrl->tickMarquee();
        vTaskDelay(pdMS_TO_TICKS(400));
    }
}
```

```cpp
#include <BLEDevice.h>

#include "OTAUpdate.h"
#include "OTABleService.h"
#include "version.h"
#include "secrets.h"
#include "AiNotifyController.h"

static const int           MQTT_PORT         = 8883;
static const char          MQTT_TOPIC[]      = "ai-notify/events";
static const unsigned long MQTT_RECONNECT_MS = 5000;

static WiFiClientSecure   secureClient;
static PubSubClient       mqttClient(secureClient);
static AiNotifyController controller;

static WifiConnect    wifi;
static WifiBleService wifiBleService;
static OTAUpdate      otaUpdate;
static OTABleService  otaBleService;

static BLEServer*  pServer           = nullptr;
static bool        deviceConnected   = false;
static bool        oldDeviceConnected = false;
static bool        sdCardReady       = false;
static unsigned long lastMqttReconnect = 0;

class ServerCallbacks : public BLEServerCallbacks {
    void onConnect(BLEServer*)    override { deviceConnected = true; }
    void onDisconnect(BLEServer*) override { deviceConnected = false; }
};

static void mqttCallback(char* topic, byte* payload, unsigned int length) {
    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, payload, length);
    if (err) {
        LOG_E("MQTT JSON parse error");
        return;
    }

    const char* event = doc["event"] | "";
    if (strcmp(event, "question") == 0) {
        controller.onQuestion(String(doc["text"] | ""));
    } else if (strcmp(event, "ai_start") == 0) {
        controller.onAiStart();
    } else if (strcmp(event, "tool_start") == 0) {
        controller.onToolStart(String(doc["tool"] | ""));
    } else if (strcmp(event, "tool_end") == 0) {
        controller.onToolEnd(String(doc["tool"] | ""));
    } else if (strcmp(event, "done") == 0) {
        controller.onDone();
    }
}

static bool mqttConnect() {
    secureClient.setCACert(AWS_CERT_CA);
    secureClient.setCertificate(AWS_CERT_CRT);
    secureClient.setPrivateKey(AWS_CERT_PRIVATE);
    mqttClient.setServer(AWS_IOT_ENDPOINT, MQTT_PORT);
    mqttClient.setCallback(mqttCallback);
    mqttClient.setBufferSize(4096);   // question text can be up to 2000 chars

    LOG_IF("Connecting MQTT to %s ...", AWS_IOT_ENDPOINT);
    if (mqttClient.connect(AWS_IOT_THING_NAME)) {
        mqttClient.subscribe(MQTT_TOPIC);
        LOG_IF("MQTT subscribed: %s", MQTT_TOPIC);
        return true;
    }
    LOG_IF("MQTT connect failed, rc=%d", mqttClient.state());
    return false;
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("AI Notify starting");

    sdCardReady = SDLogger::getInstance().init("/logs");
    if (sdCardReady) {
        LOG_I("=== AI Notify ===");
        LOG_IF("Version: %s", FIRMWARE_VERSION);
        LOG_IF("Build: %s %s", __DATE__, __TIME__);
    }

    controller.begin();

    xTaskCreate(ledRainbowTask, "ledRainbow", 2048, &controller, 1, nullptr);
    xTaskCreate(lcdMarqueeTask, "lcdMarquee", 2048, &controller, 1, nullptr);

    BLEDevice::init("AiNotify");
    pServer = BLEDevice::createServer();
    pServer->setCallbacks(new ServerCallbacks());

    wifiBleService.setWifi(&wifi);
    wifiBleService.init(pServer);

    otaBleService.setOTAUpdate(&otaUpdate);
    otaBleService.setWifi(&wifi);
    otaBleService.setVersionString(FIRMWARE_VERSION);
    otaBleService.init(pServer);

    BLEDevice::getAdvertising()->start();
    LOG_I("BLE advertising started");

    wifi.begin();
    if (wifi.hasCredentials()) {
        LOG_IF("Connecting to WiFi '%s'...", wifi.getSSID().c_str());
        if (wifi.connect() == 0) {
            LOG_IF("WiFi connected: %s", WiFi.localIP().toString().c_str());
            mqttConnect();
        } else {
            LOG_W("WiFi connection failed — configure via BLE");
        }
    } else {
        LOG_I("No WiFi credentials — configure via BLE");
    }

    LOG_I("Setup complete");
}

void loop() {
    wifiBleService.handle();
    otaBleService.handle();

    // BLE reconnect advertising
    if (deviceConnected && !oldDeviceConnected) {
        oldDeviceConnected = deviceConnected;
    }
    if (!deviceConnected && oldDeviceConnected) {
        delay(500);
        BLEDevice::getAdvertising()->start();
        oldDeviceConnected = deviceConnected;
    }

    // MQTT keepalive and reconnect
    bool wifiOk = (WiFi.status() == WL_CONNECTED);
    if (wifiOk) {
        if (!mqttClient.connected()) {
            unsigned long now = millis();
            if (now - lastMqttReconnect > MQTT_RECONNECT_MS) {
                lastMqttReconnect = now;
                if (!mqttConnect()) {
                    controller.onWifiReconnecting();
                }
            }
        } else {
            mqttClient.loop();
        }
    } else {
        controller.onWifiReconnecting();
    }

    delay(10);
}
```

```typescript
// --- AI Notify: publish MQTT events alongside SSE ---
// Uses @aws-sdk/client-iot-data-plane which is available in the Lambda Node 20 runtime
// without any additional npm install.
import { IoTDataPlaneClient, PublishCommand } from '@aws-sdk/client-iot-data-plane';

let iotClient: IoTDataPlaneClient | null = null;

async function getIotClient(): Promise<IoTDataPlaneClient> {
  if (iotClient) return iotClient;
  const result = await ssmClient.send(new GetParameterCommand({ Name: '/AiNotify/IotEndpoint' }));
  const endpoint = result.Parameter?.Value;
  if (!endpoint) throw new Error('/AiNotify/IotEndpoint not found in SSM');
  iotClient = new IoTDataPlaneClient({ endpoint: `https://${endpoint}` });
  return iotClient;
}

async function publishAiNotify(payload: object): Promise<void> {
  try {
    const client = await getIotClient();
    await client.send(new PublishCommand({
      topic: 'ai-notify/events',
      payload: Buffer.from(JSON.stringify(payload)) as unknown as Uint8Array,
      qos: 0,
    }));
  } catch (err) {
    console.error('[ai-notify] publish failed (non-fatal):', err);
    iotClient = null;  // reset so next invocation re-fetches endpoint
  }
}
```
